estructura

src/
  index.ts          # arranca el servidor
  .env              # variables de entorno importantes antes de montar el backend hay que crearla
  app.ts            # configuración de Express
  config/
    env.ts          # variables de entorno
  db/
    pool.ts         # conexión con PostgreSQL
  routes/
    index.ts        # rutas principales
    obra.routes.ts  # rutas de la entidad Obra
  controllers/
    obra.controller.ts  #controlador de consultas, insert, delete y updates
  models/
    obra.model.ts       #variables de obras
  middlewares/
    errorHandler.ts    # manejo de errores

*****IMPORTANTE***

    .env:

    PORT=5000
    CORS_ORIGINS=http://localhost:5173,http://localhost:5174
    DB_HOST=localhost
    DB_PORT=5432
    DB_NAME=bd_museo  (El nombre de la base de datos )
    DB_USER=postgres
    DB_PASSWORD=123456 (tu contraseña de postgress)
